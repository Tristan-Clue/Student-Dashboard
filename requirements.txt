# ============================================================

# Python version: 3.11+

# ------------------------------------------------------------
# 1. FRONTEND / UI
# ------------------------------------------------------------

streamlit==1.35.0
# Core framework — renders the entire web UI.
# Handles routing, session state, file uploads, and widgets.


# ------------------------------------------------------------
# 2. DATABASE — ORM & SQLite driver
# ------------------------------------------------------------

sqlalchemy==2.0.30
# SQLAlchemy ORM: defines Python classes as database tables.
# Handles all SQL queries without writing raw SQL.
# SQLite driver is included in Python's standard library —
# no extra install needed for local development.


# ------------------------------------------------------------
# 3. AUTHENTICATION & SECURITY
# ------------------------------------------------------------

bcrypt==4.1.3
# Industry-standard password hashing library.
# Automatically salts passwords and is intentionally slow
# to resist brute-force attacks.

# Note: DO NOT store plain-text passwords. Ever.


# ------------------------------------------------------------
# 4. AI INTEGRATION
# ------------------------------------------------------------

google-generativeai==0.7.2
# Official Google Gemini Python SDK.
# Used for: Gemini chat completions (summaries, flashcards,
# key concepts extraction).
# Reads API key from environment variable GEMINI_API_KEY.


# ------------------------------------------------------------
# 5. DOCUMENT PROCESSING
# ------------------------------------------------------------

pdfplumber==0.11.0
# Primary PDF parser. More accurate than PyPDF2 for
# multi-column layouts, tables, and complex PDFs.
# Handles multi-page documents cleanly.

pypdf==4.2.0
# Fallback PDF parser — used when pdfplumber fails.
# Lighter weight, good for simple text-only PDFs.

python-docx==1.1.2
# Optional DOCX support.
# Allows users to upload Microsoft Word documents
# and extract their text content.


# ------------------------------------------------------------
# 6. ENVIRONMENT & CONFIGURATION
# ------------------------------------------------------------

python-dotenv==1.0.1
# Loads environment variables from a .env file during
# local development. In production (Streamlit Cloud),
# secrets are loaded via st.secrets instead.


# ------------------------------------------------------------
# 7. DATA HANDLING & UTILITIES
# ------------------------------------------------------------

pandas==2.2.2
# Used for tabular data display in the dashboard
# (study statistics, upload history tables).

# text to the API, ensuring chunks stay within model limits

# 8. DATE & TIME
